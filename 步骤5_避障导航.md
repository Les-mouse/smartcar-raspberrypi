## 5.1 IMU 驱动（GY-85）

### Arduino 端添加 IMU 读取

在 `car_firmware.ino` 中添加以下代码：

```cpp
// ===== 添加在文件开头 =====
#include <Adafruit_MPU6050.h>
#include <Adafruit_HMC5883L.h>

Adafruit_MPU6050 mpu;
Adafruit_HMC5883L mag;
bool imu_ok = false;

// 在 setup() 中添加：
void initIMU() {
  if (mpu.begin()) {
    mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
    mpu.setGyroRange(MPU6050_RANGE_500_DEG);
    mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);
    imu_ok = true;
    Serial.println("MPU6050 OK");
  }
  if (mag.begin()) {
    Serial.println("HMC5883L OK");
  }
}

// IMU 融合（简化版互补滤波）
float gyro_z_offset = 0;
unsigned long last_imu_time = 0;
float heading_raw = 0;

void readIMU() {
  if (!imu_ok) return;
  
  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);
  
  unsigned long now = millis();
  float dt = (now - last_imu_time) / 1000.0;
  last_imu_time = now;
  if (dt > 1.0 || dt <= 0) return;
  
  // 陀螺仪积分
  heading_raw += (g.gyro.z - gyro_z_offset) * dt;
  
  // 磁力计修正（每10次修正一次）
  static int mag_counter = 0;
  mag_counter++;
  if (mag_counter > 10 && mag.begin()) {
    sensors_event_t mag_event;
    mag.getEvent(&mag_event);
    float mag_heading = atan2(mag_event.magnetic.y, 
                              mag_event.magnetic.x) * 180.0 / PI;
    
    // 互补滤波
    heading = 0.98 * heading_raw + 0.02 * mag_heading;
    heading_raw = heading;
    mag_counter = 0;
  }
  
  heading = heading_raw;
}

// 陀螺仪校准（在 setup 中调用，车静止时）
void calibrateGyro() {
  float sum = 0;
  for (int i = 0; i < 200; i++) {
    sensors_event_t a, g, temp;
    mpu.getEvent(&a, &g, &temp);
    sum += g.gyro.z;
    delay(5);
  }
  gyro_z_offset = sum / 200.0;
}

// ===== 在 setup() 末尾添加 =====
initIMU();
calibrateGyro();

// ===== 在 loop() 中取消注释 =====
readIMU();
```

# ~/ros2_ws/src/smartcar_nav/smartcar_nav/avoidance.py
"""
简易避障：读超声波 → 计算安全速度 → 发布 /cmd_vel
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist
from sensor_msgs.msg import Range
from std_msgs.msg import String
import json
import math

class AvoidanceNode(Node):
    def __init__(self):
        super().__init__('avoidance')
        
        # 超声波距离（前后左右）
        self.distances = {'front': 999, 'back': 999, 
                          'left': 999, 'right': 999}
        
        # 参数
        self.safe_distance = 0.3  # 安全距离 30cm
        self.max_speed = 0.3      # 最大线速度 m/s
        
        # 发布速度指令
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        
        # 订阅超声波
        self.create_subscription(Range, '/ultrasonic/front', 
            lambda m: self.us_callback('front', m), 10)
        self.create_subscription(Range, '/ultrasonic/back', 
            lambda m: self.us_callback('back', m), 10)
        self.create_subscription(Range, '/ultrasonic/left', 
            lambda m: self.us_callback('left', m), 10)
        self.create_subscription(Range, '/ultrasonic/right', 
            lambda m: self.us_callback('right', m), 10)
        
        # 订阅检测结果（用于目标跟踪）
        self.create_subscription(String, '/camera/detections',
            self.detection_callback, 10)
        
        # 当前模式
        self.mode = 'patrol'  # patrol / track / stop
        self.target_object = None
        self.last_detection = None
        
        # 控制循环
        self.timer = self.create_timer(0.1, self.control_loop)
        
        self.get_logger().info('避障节点就绪')
    
    def us_callback(self, direction, msg):
        self.distances[direction] = msg.range
    
    def detection_callback(self, msg):
        try:
            detections = json.loads(msg.data)
            self.last_detection = detections
        except:
            pass
    
    def control_loop(self):
        """主控制循环"""
        if self.mode == 'stop':
            self.stop()
            return
        
        cmd = Twist()
        
        if self.mode == 'patrol':
            # 巡逻模式：前方无障碍就前进，有障碍就转向
            front = self.distances['front']
            left = self.distances.get('left', 999)
            right = self.distances.get('right', 999)
            
            if front > self.safe_distance:
                # 安全，前进
                cmd.linear.x = self.max_speed
                cmd.angular.z = 0.0
            else:
                # 有障碍，选择空旷方向
                if left > right:
                    cmd.angular.z = 0.5  # 左转
                    cmd.linear.x = 0.05  # 慢速
                    self.get_logger().info('左转绕行')
                else:
                    cmd.angular.z = -0.5  # 右转
                    cmd.linear.x = 0.05
                    self.get_logger().info('右转绕行')
        
        elif self.mode == 'track' and self.last_detection:
            # 跟踪模式：保持目标在画面中央
            detections = self.last_detection
            target = None
            
            # 找置信度最高的目标
            for d in detections:
                if d['class'] == self.target_object or self.target_object is None:
                    if target is None or d['confidence'] > target['confidence']:
                        target = d
            
            if target:
                # 计算物体在画面中的位置 (0=左, 640=右)
                cx = target['x']
                error = (cx - 320) / 320.0  # -1到1
                
                cmd.angular.z = -error * 0.3  # 物体在左边就往左转
                cmd.linear.x = self.max_speed * (1.0 - abs(error))
                
                # 物体太大说明太近，后退
                if target['width'] > 200:
                    cmd.linear.x = -0.1
                
                self.get_logger().info(
                    f"跟踪 {target.get('class', 'object')} "
                    f"error={error:.2f} size={target['width']}")
            else:
                cmd.angular.z = 0.3  # 找不到目标，慢慢转圈找
        
        self.cmd_pub.publish(cmd)
    
    def stop(self):
        cmd = Twist()
        self.cmd_pub.publish(cmd)

def main():
    rclpy.init()
    node = AvoidanceNode()
    rclpy.spin(node)
    node.destroy_node()
    rclpy.shutdown()
```

# ~/ros2_ws/src/smartcar_nav/smartcar_nav/odometry.py
"""
从编码器 + IMU 推算小车位姿
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist, Pose2D
from std_msgs.msg import String
import json
import math
import threading

class OdometryNode(Node):
    def __init__(self):
        super().__init__('odometry')
        
        self.x = 0.0
        self.y = 0.0
        self.theta = 0.0
        
        self.last_enc = [0, 0, 0, 0]
        self.ticks_per_meter = 1000  # 需要实际标定
        
        self.pose_pub = self.create_publisher(Pose2D, '/pose', 10)
        self.create_subscription(String, '/arduino/telemetry', 
            self.telem_callback, 10)
        
        self.get_logger().info('里程计就绪')
    
    def telem_callback(self, msg):
        try:
            data = json.loads(msg.data)
            enc = data.get('enc', [0,0,0,0])
            
            # 计算位移
            avg_delta = 0
            for i in range(4):
                delta = enc[i] - self.last_enc[i]
                avg_delta += delta
            avg_delta /= 4.0
            
            # 前进距离 (m)
            distance = avg_delta / self.ticks_per_meter
            
            # 更新位姿
            self.x += distance * math.cos(self.theta)
            self.y += distance * math.sin(self.theta)
            
            # 航向角 (从 Arduino 传来的 IMU 数据)
            if 'heading' in data:
                self.theta = math.radians(data['heading'])
            
            self.last_enc = enc
            
            # 发布位姿
            pose = Pose2D()
            pose.x = self.x
            pose.y = self.y
            pose.theta = self.theta
            self.pose_pub.publish(pose)
            
        except Exception as e:
            pass

def main():
    rclpy.init()
    node = OdometryNode()
    rclpy.spin(node)
```

## 5.4 航点巡逻

```python
# ~/ros2_ws/src/smartcar_nav/smartcar_nav/patrol.py
"""
航点巡逻：依次前往预设航点
"""

import rclpy
from rclpy.node import Node
from geometry_msgs.msg import Twist, Pose2D
import math

class PatrolNode(Node):
    def __init__(self):
        super().__init__('patrol')
        
        # 预设巡逻路线 (x, y) 单位：米
        self.waypoints = [
            (1.0, 0.0),
            (1.0, 1.0),
            (0.0, 1.0),
            (0.0, 0.0),  # 回到原点
        ]
        self.current_waypoint = 0
        
        # 当前位姿
        self.pose = Pose2D()
        self.create_subscription(Pose2D, '/pose', self.pose_callback, 10)
        
        # 发布速度
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        
        # 控制循环
        self.timer = self.create_timer(0.1, self.navigate)
        
        self.arrival_threshold = 0.15  # 到达阈值 15cm
        
        self.get_logger().info('巡逻节点就绪')
    
    def pose_callback(self, msg):
        self.pose = msg
    
    def navigate(self):
        if self.current_waypoint >= len(self.waypoints):
            self.get_logger().info('巡逻完成！回到起点')
            self.current_waypoint = 0
        
        target = self.waypoints[self.current_waypoint]
        
        # 计算到目标的方向和距离
        dx = target[0] - self.pose.x
        dy = target[1] - self.pose.y
        distance = math.sqrt(dx*dx + dy*dy)
        
        # 目标角度
        target_angle = math.atan2(dy, dx)
        angle_error = target_angle - self.pose.theta
        # 归一化到 [-PI, PI]
        angle_error = math.atan2(math.sin(angle_error), 
                                 math.cos(angle_error))
        
        cmd = Twist()
        
        if distance < self.arrival_threshold:
            # 到达，切换下一航点
            self.get_logger().info(
                f'到达航点{self.current_waypoint} '
                f'({target[0]:.1f}, {target[1]:.1f})')
            self.current_waypoint += 1
            cmd.linear.x = 0.0
            cmd.angular.z = 0.0
        else:
            # 先转向（角度误差 > 0.2 rad）
            if abs(angle_error) > 0.2:
                cmd.angular.z = 0.3 * (1 if angle_error > 0 else -1)
                cmd.linear.x = 0.05  # 慢速转向
            else:
                # 面向目标，前进
                cmd.angular.z = angle_error * 0.5  # 微小角度修正
                cmd.linear.x = min(0.3, distance * 0.5)  # 越近越慢
        
        self.cmd_pub.publish(cmd)

def main():
    rclpy.init()
    node = PatrolNode()
    rclpy.spin(node)
```
