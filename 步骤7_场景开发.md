## 7.1 场景1：家庭巡逻 + 异常报告

```python
# ~/robot_bridge/scenes/patrol_scene.py
"""
巡逻场景：沿路线走，YOLO检测异常 → 拍照 → 报告
"""

import json
import time
import subprocess

class PatrolScene:
    def __init__(self, bridge):
        self.bridge = bridge
        self.route = [
            {'name': '客厅', 'x': 0, 'y': 0},
            {'name': '走廊', 'x': 2, 'y': 0},
            {'name': '卧室门口', 'x': 2, 'y': 1.5},
            {'name': '厨房', 'x': 0, 'y': 1.5},
        ]
        self.alerts = []
    
    def run(self):
        """执行巡逻"""
        results = []
        
        for waypoint in self.route:
            print(f"[巡逻] 前往 {waypoint['name']}...")
            
            # 到达航点
            # self.bridge.cmd_move({'distance': waypoint['distance']})
            time.sleep(2)  # 模拟移动时间
            
            # 360° 扫描
            print(f"[巡逻] 在 {waypoint['name']} 扫描...")
            scan_result = self.bridge.cmd_scan({})
            
            # 检查异常
            objects = scan_result.get('objects', [])
            anomalies = self.check_anomalies(objects, waypoint['name'])
            
            if anomalies:
                # 拍照存证
                snap = self.bridge.cmd_snap({})
                results.append({
                    'location': waypoint['name'],
                    'anomalies': anomalies,
                    'image': snap.get('image_url', ''),
                    'time': time.strftime('%H:%M:%S')
                })
        
        return self.summarize(results)
    
    def check_anomalies(self, objects, location):
        """检测异常情况"""
        anomalies = []
        
        obj_names = [o.get('class', '') for o in objects]
        
        # 规则1: 检测到人（但你在家不算异常，这里简化）
        if '人' in obj_names and location == '走廊':
            anomalies.append('走廊有人经过')
        
        # 规则2: 地上有不应该出现的东西
        floor_items = ['书', '瓶子', '杯子', '手机', '遥控器']
        for item in floor_items:
            if item in obj_names:
                anomalies.append(f'发现{item}在地上')
        
        # 规则3: 门开着（需要自定义训练）
        # 规则4: 灯没关（需要光敏传感器）
        
        return anomalies
    
    def summarize(self, results):
        """生成巡逻报告"""
        report = "=== 巡逻报告 ===\n"
        report += f"时间: {time.strftime('%Y-%m-%d %H:%M')}\n"
        report += f"巡查点位: {len(self.route)} 处\n"
        
        if not results:
            report += "✅ 一切正常\n"
        else:
            for r in results:
                report += f"\n📍 {r['location']} ({r['time']})\n"
                for a in r['anomalies']:
                    report += f"  ⚠️ {a}\n"
                if r.get('image'):
                    report += f"  📷 照片: {r['image']}\n"
        
        return report

# 测试
if __name__ == '__main__':

    bridge = RobotBridge()
    scene = PatrolScene(bridge)
    report = scene.run()
    print(report)
```

## 7.2 场景2：找东西

```python
# ~/robot_bridge/scenes/find_object.py
"""
找东西场景：全屋巡逻，YOLO检测指定物体 → 找到停止并报告位置
"""

import time

class FindObjectScene:
    def __init__(self, bridge):
        self.bridge = bridge
        self.found = False
    
    def run(self, target):
        """寻找指定物体"""
        print(f"[找东西] 开始寻找: {target}")
        self.bridge.cmd_say({'text': f'好的，我现在去找{target}'})
        
        # 策略：先原地转一圈扫描
        scan = self.bridge.cmd_scan({})
        for obj in scan.get('objects', []):
            if obj.get('class') == target and obj.get('confidence', 0) > 0.6:
                snap = self.bridge.cmd_snap({})
                return {
                    'found': True,
                    'object': target,
                    'location': '当前视野中',
                    'confidence': obj['confidence'],
                    'image': snap.get('image_url', '')
                }
        
        # 视野中没有，启动巡逻寻找
        # 每到一个位置就 scan，直到找到或巡逻完
        
        # 简化：巡逻过程中持续 YOLO 检测
        patrol_positions = [
            {'x': 0, 'y': 0, 'name': '起点'},
            {'x': 1, 'y': 0, 'name': '客厅中央'},
            {'x': 2, 'y': 0, 'name': '走廊'},
            {'x': 2, 'y': 1, 'name': '卧室前'},
            {'x': 0, 'y': 1, 'name': '厨房'},
        ]
        
        for pos in patrol_positions:
            result = self.bridge.cmd_move({'distance': 0.5})  # 移动
            time.sleep(2)
            
            scan = self.bridge.cmd_scan({})
            for obj in scan.get('objects', []):
                if obj.get('class') == target and obj.get('confidence', 0) > 0.5:
                    self.bridge.cmd_stop({})
                    snap = self.bridge.cmd_snap({})
                    self.bridge.cmd_say({'text': f'找到了！{target}在{pos["name"]}'})
                    return {
                        'found': True,
                        'object': target,
                        'location': pos['name'],
                        'confidence': obj['confidence'],
                        'image': snap.get('image_url', '')
                    }
            
            print(f"  [{pos['name']}] 未找到 {target}")
        
        # 没找到
        self.bridge.cmd_say({'text': f'抱歉，没有找到{target}'})
        return {
            'found': False,
            'object': target,
            'message': f'巡逻完毕，未发现{target}'
        }
```

## 7.3 场景3：定点配送

```python
# ~/robot_bridge/scenes/delivery.py
"""
配送场景：从一个位置导航到另一个位置
需要预先标定好房间坐标
"""

import time

class DeliveryScene:
    def __init__(self, bridge):
        self.bridge = bridge
        
        # 预标定的房间坐标（需要实际测量）
        self.rooms = {
            '客厅': {'x': 0, 'y': 0},
            '书房': {'x': 3, 'y': 0},
            '卧室': {'x': 3, 'y': 2},
            '厨房': {'x': 0, 'y': 2},
        }
    
    def deliver(self, from_room, to_room):
        """从 from_room 送到 to_room"""
        
        if to_room not in self.rooms:
            return {'error': f'未知房间: {to_room}，已知: {list(self.rooms.keys())}'}
        
        target = self.rooms[to_room]
        
        self.bridge.cmd_say({'text': f'好的，我现在送去{to_room}'})
        
        # 简单导航：面向目标方向 → 前进
        # (精确实现需要 Nav2 + SLAM)
        
        # 估算距离和角度
        import math
        dx = target['x']
        dy = target['y']
        distance = math.sqrt(dx*dx + dy*dy)
        angle = math.degrees(math.atan2(dy, dx))
        
        if abs(angle) > 5:
            self.bridge.cmd_rotate({'angle': angle})
            time.sleep(2)
        
        self.bridge.cmd_move({'distance': distance, 'speed': 0.2})
        time.sleep(distance / 0.2 + 1)
        
        self.bridge.cmd_say({'text': f'已到达{to_room}，请取货'})
        
        # 拍照确认
        snap = self.bridge.cmd_snap({})
        
        return {
            'status': 'delivered',
            'to': to_room,
            'image': snap.get('image_url', '')
        }
```

## 7.4 整合到 Bridge Server

修改 `bridge_server.py`，添加场景调用：

```python
# 在 RobotBridge 类中添加：

from scenes.patrol_scene import PatrolScene
from scenes.find_object import FindObjectScene  
from scenes.delivery import DeliveryScene

class RobotBridge:
    def __init__(self):
        # ... 原有代码 ...
        self.scenes = {
            'patrol': PatrolScene(self),
            'find': FindObjectScene(self),
            'deliver': DeliveryScene(self),
        }
    
    def execute_command(self, action, params):
        cmd_map = {
            'move': self.cmd_move,
            'rotate': self.cmd_rotate,
            'stop': self.cmd_stop,
            'snap': self.cmd_snap,
            'scan': self.cmd_scan,
            'say': self.cmd_say,
            # 场景
            'patrol_full': self.scene_patrol,
            'find_object': self.scene_find,
            'deliver_to': self.scene_deliver,
        }
        func = cmd_map.get(action)
        if func:
            return func(params)
        return {'error': f'未知动作: {action}'}
    
    def scene_patrol(self, params):
        return self.scenes['patrol'].run()
    
    def scene_find(self, params):
        target = params.get('target', '手机')
        return self.scenes['find'].run(target)
    
    def scene_deliver(self, params):
        to_room = params.get('to', '书房')
        from_room = params.get('from', '客厅')
        return self.scenes['deliver'].deliver(from_room, to_room)
```
